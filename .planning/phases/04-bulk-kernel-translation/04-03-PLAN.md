Concrete approach — write a small one-off enumeration script `tools/generate_gga_roster.py`:

```python
#!/usr/bin/env python3
"""One-shot: scan crates/kernel-gga-*/src/**/mod.rs and print a GgaFunctional enum roster."""
import re
from pathlib import Path

WORKSPACE = Path(__file__).resolve().parent.parent
GGA_CRATES = sorted(WORKSPACE.glob("crates/kernel-gga-*"))

# Load libxc IDs from src/meta/generated.rs (regex for `id: FunctionalId(N),` near `name: "gga_x_pbe",`).
meta_text = (WORKSPACE / "src/meta/generated.rs").read_text()
ID_MAP = {m.group(2): int(m.group(1))
          for m in re.finditer(r'id:\s*FunctionalId\((\d+)\)[^}]+?name:\s*"([a-z0-9_]+)"', meta_text, re.DOTALL)}

# Scalar-arg extractor: match the signature between `&mut Array<f64>,` output and trailing `dens_threshold,`.
SIG_RE = re.compile(
    r'#\[cube\(launch_unchecked\)\][\s\S]+?pub fn\s+\w+\(([\s\S]+?)\)\s*\{',
    re.MULTILINE,
)

def extract_scalars(rs_text: str) -> list[str]:
    # Split signature body, strip rho/sigma/zk/v*/lapl/tau, strip the trailing thresholds, return middle scalars.
    m = SIG_RE.search(rs_text)
    if not m:
        return []
    body = m.group(1)
    scalars = []
    for arg in [a.strip() for a in body.split(",") if a.strip()]:
        # arg looks like "rho: &Array<f64>" or "param_BB: f64" or "dens_threshold: f64"
        if ":" not in arg:
            continue
        name, ty = [s.strip() for s in arg.split(":", 1)]
        if ty == "f64" and name not in ("dens_threshold", "zeta_threshold"):
            scalars.append(name)
    return scalars

for crate_dir in GGA_CRATES:
    batch = crate_dir.name.replace("kernel-gga-", "")   # "1a", "22", etc.
    lib_rs = crate_dir / "src/lib.rs"
    if not lib_rs.exists(): continue
    functionals = re.findall(r'pub mod (gga_[a-z0-9_]+);', lib_rs.read_text())
    for fn in functionals:
        func_dir = crate_dir / "src" / fn
        if not func_dir.is_dir():
            continue
        files = {p.stem for p in func_dir.glob("*.rs")} - {"mod"}
        has_exc = "exc_unpol" in files or "exc_pol" in files
        libxc_id = ID_MAP.get(fn, "UNKNOWN")
        # Extract per-functional scalars from the first available kernel file
        sig_source = None
        for cand in ("exc_unpol.rs", "vxc_unpol.rs", "exc_pol.rs", "vxc_pol.rs"):
            p = func_dir / cand
            if p.exists():
                sig_source = p.read_text()
                break
        scalars = extract_scalars(sig_source) if sig_source else []
        print(f"{fn}\t{batch}\t{libxc_id}\t{has_exc}\t{','.join(scalars)}")
```

///
/// Variants are ordered by libxc ID for consistency with the registry.
/// Non-compiled GGA functionals from `src/meta/generated.rs` are NOT represented
/// here — `GgaFunctional::from_id` returns `Err(UnsupportedFunctional)` for them.

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum GgaFunctional {
    // Sorted by libxc id:
    GgaXGam,            // id 32
    GgaCGam,            // id 33
    GgaXHcthA,          // id 34
    // ... 103 more ...
    GgaXLb,             // id 160 (vxc-only)
    // ...
    GgaXQ1d,            // id 734 (last by id)

}

use crate::dims::Dimensions;
use crate::error::LibxcRsError;
use crate::input::GgaInput;
use crate::kernel::launch::{cpu_client, create_input_buffer, create_zero_output_buffer, calculate_launch_config};
use crate::model::{DerivativeOrder, GgaFunctional, Thresholds};
use crate::output::GgaOutput;

pub fn dispatch_gga(
    functional: GgaFunctional,
    input: &GgaInput,
    order: DerivativeOrder,
    output: &mut GgaOutput,
    thresholds: &Thresholds,
) -> Result<(), LibxcRsError> {
    let np = input.np();
    let spin = input.spin();
    let dims = Dimensions::gga(spin);

    // 1. Validate order vs functional capability
    if order == DerivativeOrder::Exc && !functional.has_exc() {
        return Err(LibxcRsError::UnsupportedDerivativeOrder {
            id: functional_id_of_gga(functional),
            order,
            max: DerivativeOrder::Vxc,
        });
    }

    // 2. Zero caller output buffers (all 15 Option<&mut [f64]> fields)
    for field in [&mut output.zk, &mut output.vrho, &mut output.vsigma,
                  &mut output.v2rho2, &mut output.v2rhosigma, &mut output.v2sigma2,
                  &mut output.v3rho3, &mut output.v3rho2sigma, &mut output.v3rhosigma2, &mut output.v3sigma3,
                  &mut output.v4rho4, &mut output.v4rho3sigma, &mut output.v4rho2sigma2,
                  &mut output.v4rhosigma3, &mut output.v4sigma4] {
        if let Some(buf) = field.as_deref_mut() { buf.fill(0.0); }
    }

    // 3. Build client + input handles
    let client = cpu_client();
    let rho_handle = create_input_buffer(&client, input.rho());
    let rho_len = input.rho().len();
    let sigma_handle = create_input_buffer(&client, input.sigma());
    let sigma_len = input.sigma().len();

    // 4. Create output handles sized by Dimensions::gga(spin) for each derivative level required by `order`.
    //    (15 Option<Handle>s, gated by order.)
    // ... zk_handle, vrho_handle, vsigma_handle, v2*_handle, v3*_handle, v4*_handle ...

    let (cube_count, cube_dim) = calculate_launch_config(np);

    // 5. Route to the per-batch launch helper. Each GgaFunctional variant lives
    //    in exactly one batch (TSV column 2 from Task 1). Outer match:
    match functional {
        GgaFunctional::GgaCFt97 => batch1a::dispatch_gga_c_ft97_launch(/* handles + lens + order + spin + thresholds */)?,
        GgaFunctional::GgaXPbe  => batch11::dispatch_gga_x_pbe_launch(/* same */)?,
        GgaFunctional::GgaCPbe  => batch4g::dispatch_gga_c_pbe_launch(/* same */)?,
        // ... 103 more ...
        GgaFunctional::GgaXLb   => batch8d::dispatch_gga_x_lb_launch(/* vxc-only */)?,
    }

    // 6. Read back results into caller slices (15 fields)
    Ok(())
}
