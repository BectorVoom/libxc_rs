Mismatches between the two (e.g., Rust has exc_* files but libxc flag is unset, or vice versa) are legitimate: just skip the comparison when either side says no.

```python
#!/usr/bin/env python3
"""One-shot: scan crates/kernel-mgga-*/src/**/mod.rs and print an MggaFunctional enum roster."""
import re
from pathlib import Path

WORKSPACE = Path(__file__).resolve().parent.parent
MGGA_CRATES = [c for c in sorted(WORKSPACE.glob("crates/kernel-mgga-*")) if c.name != "kernel-mgga"]

meta_text = (WORKSPACE / "src/meta/generated.rs").read_text()
ID_MAP = {m.group(2): int(m.group(1))
          for m in re.finditer(r'id:\s*FunctionalId\((\d+)\)[^}]+?name:\s*"([a-z0-9_]+)"', meta_text, re.DOTALL)}

# Scalar extractor — same shape as tools/generate_gga_roster.py.
SIG_RE = re.compile(r'#\[cube\(launch_unchecked\)\][\s\S]+?pub fn\s+\w+\(([\s\S]+?)\)\s*\{', re.MULTILINE)
def extract_scalars(rs_text: str) -> list[str]:
    m = SIG_RE.search(rs_text)
    if not m: return []
    out = []
    for arg in [a.strip() for a in m.group(1).split(",") if a.strip()]:
        if ":" not in arg: continue
        name, ty = [s.strip() for s in arg.split(":", 1)]
        if ty == "f64" and name not in ("dens_threshold", "zeta_threshold"):
            out.append(name)
    return out

compiled = []
for crate_dir in MGGA_CRATES:
    batch = crate_dir.name.replace("kernel-mgga-", "")
    lib_rs = crate_dir / "src/lib.rs"
    if not lib_rs.exists():
        continue
    for line in lib_rs.read_text().splitlines():
        stripped = line.strip()
        if stripped.startswith("//"):
            continue  # Skip commented-out (deferred) modules
        m = re.match(r'pub mod (mgga_[a-z0-9_]+);', stripped)
        if not m:
            continue
        fn = m.group(1)
        func_dir = crate_dir / "src" / fn
        if not func_dir.is_dir():
            continue
        files = {p.stem for p in func_dir.glob("*.rs")} - {"mod"}
        has_exc = "exc_unpol" in files or "exc_pol" in files
        libxc_id = ID_MAP.get(fn, "UNKNOWN")
        sig_source = None
        for cand in ("exc_unpol.rs", "vxc_unpol.rs", "exc_pol.rs", "vxc_pol.rs"):
            p = func_dir / cand
            if p.exists():
                sig_source = p.read_text()
                break
        scalars = extract_scalars(sig_source) if sig_source else []
        compiled.append((fn, batch, libxc_id, has_exc, ",".join(scalars)))

for row in compiled:
    print("\t".join(str(x) for x in row))
print(f"# total_compiled: {len(compiled)}")
```

```rust
use crate::error::LibxcRsError;
use crate::model::FunctionalId;
use libxc_kernel_mgga::deferred::is_deferred as is_deferred_mgga;

/// Enumerates the 86 compiled MGGA functionals dispatch_mgga can route.
///
/// Variants are sorted by libxc ID. Deferred functionals (6 IDs tracked in
/// `crates/kernel-mgga/src/deferred.rs`) are NOT represented here —
/// `MggaFunctional::from_id` returns `UnsupportedFunctional` for them.
///
/// `_vxc`-only functionals lack an exc kernel; `has_exc()` returns false for them
/// based on filesystem presence of exc_*.rs modules, NOT on oracle flags (W5).
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum MggaFunctional {
    MggaCDldf,              // id 37
    MggaXcZlp,              // id 42
    // ... 84 more, from mgga_roster.tsv ...
}

impl MggaFunctional {
    /// Map a libxc functional ID to a dispatchable MGGA variant.
    pub fn from_id(id: FunctionalId) -> Result<Self, LibxcRsError> {
        if is_deferred_mgga(id.raw()) {
            return Err(LibxcRsError::UnsupportedFunctional {
                id,
                reason: "MGGA functional deferred pending Brent's method root-finder. See crates/kernel-mgga/src/deferred.rs",
            });
        }
        match id.raw() {
            37 => Ok(Self::MggaCDldf),
            42 => Ok(Self::MggaXcZlp),
            // ... 84 more ...
            _ => Err(LibxcRsError::UnsupportedFunctional {
                id,
                reason: "MGGA functional not yet translated into crates/kernel-mgga*",
            }),
        }
    }

    /// True when the kernel module contains exc_unpol.rs / exc_pol.rs on disk (W5).
    ///
    /// This is a structural check independent of libxc's FLAGS_HAVE_EXC. The
    /// only two MGGA variants where Phase 4 executor confirmed via filesystem
    /// inspection that the exc_*.rs files are absent are MggaXTb09 (id 208) and
    /// MggaX2dPrp10 (id 211). Functionals like mgga_x_bj06 / mgga_x_rpp09 are
    /// NOT in MggaFunctional at all (they were never compiled); the oracle
    /// tests skip them via the `MggaFunctional::from_id -> UnsupportedFunctional`
    /// branch, not via has_exc.
    pub fn has_exc(self) -> bool {
        !matches!(self, Self::MggaX2dPrp10 | Self::MggaXTb09)
    }
}
```
