    // (compat::raw_handle::FunctionalSlot, extern_c_wrapper, c_layout::xc_func_type already imported by 06-02 setters)

    /// Helper: convert a NULL-able `*mut f64` to `Option<&mut [f64]>` of length `np * stride`.
    /// Returns None for NULL pointers (libxc "skip this derivative" semantics).
    /// SAFETY: caller asserts that non-NULL pointers point to a buffer of at least
    /// `np * stride` `f64`s, and that no aliasing exists for the returned mutable slice.
    unsafe fn ptr_to_opt_slice<'a>(ptr: *mut f64, np: usize, stride: usize) -> Option<&'a mut [f64]> {
        if ptr.is_null() { None }
        else {
            let len = np.checked_mul(stride).expect("np * stride overflow");
            // SAFETY: caller's contract.
            Some(unsafe { std::slice::from_raw_parts_mut(ptr, len) })
        }
    }

    /// Helper: build LdaInput + LdaOutput + workspace, run `evaluate_lda`, return Ok(0).
    /// All NULL output pointers map to None (skip).
    unsafe fn lda_evaluate(
        p: *const xc_func_type, np: usize, rho: *const f64, order: DerivativeOrder,
        zk: *mut f64, vrho: *mut f64, v2rho2: *mut f64, v3rho3: *mut f64, v4rho4: *mut f64,
    ) -> Result<i32, crate::LibxcRsError> {
        // SAFETY: p is non-null + initialized (wrapper macro + slot accessor enforce this).
        let f = unsafe { FunctionalSlot::as_initialized_const(p)? };
        let spin = f.spin();
