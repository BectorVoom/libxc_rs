```bash
mkdir -p log
LOG=log/09-06-spec-acceptance.log
PASS=0
FAIL=0

run_check() {
  local label="$1"; local cmd="$2"; local expect="$3"  # expect = "exit-zero" | "no-output"
  local out rc
  out=$(eval "$cmd" 2>&1)
  rc=$?
  case "$expect" in
    exit-zero)
      if [ "$rc" -eq 0 ]; then echo "PASS  $label  (exit=0)"; PASS=$((PASS+1));
      else echo "FAIL  $label  (exit=$rc) -- output: $(echo "$out" | head -3)"; FAIL=$((FAIL+1)); fi
      ;;
    no-output)
      if [ -z "$out" ]; then echo "PASS  $label  (no matches)"; PASS=$((PASS+1));
      else echo "FAIL  $label  (matches found): $(echo "$out" | head -5)"; FAIL=$((FAIL+1)); fi
      ;;
  esac
}

{
  echo "=== Plan 09-06 Task 2: SPEC §Acceptance Criteria re-verification ==="
  echo "date: $(date -Iseconds)"
  echo "git HEAD: $(git rev-parse HEAD)"
  echo
```
