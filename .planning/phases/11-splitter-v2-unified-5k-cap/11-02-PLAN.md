    ```python
    """CSE-aware compute_lines partitioner for splitter v2 (Phase 11 D-01).

    Build env source of truth: .cargo/config.toml (do not duplicate values here).
    """
    from dataclasses import dataclass

    CHUNK_MAX_LINES = 4500          # leave headroom vs the 5000 hard cap (D-LOCK-B)
    MIN_REVERSE_DEPS = 5            # natural-breakpoint reverse-dep threshold (RESEARCH heuristic)
    MIN_CHAIN_LENGTH = 50           # min def-use chain span before a breakpoint counts
    MAX_TUPLE_ARITY = 16            # cubecl-macros tuple arity cap (drop to 8 if spike-confirmed needed)

    @dataclass
    class Chunk:
        index: int                 # 0-based, sequence order (deterministic chunk id)
        lines: list[str]           # the compute lines in this chunk, in original order
        inputs: list[str]          # vars referenced inside but defined before this chunk (tuple args)
        outputs: list[str]         # vars defined inside with >=1 use after this chunk closes (tuple returns)
        est_lines: int             # estimated emitted Rust line count

    def partition_compute_lines(compute_lines, var_deps, est_line_fn, *,
                                chunk_max_lines=CHUNK_MAX_LINES) -> list[Chunk]:
        """Walk lines in order; accumulate into the current chunk until size hits
        chunk_max_lines OR a natural breakpoint (a var with reverse-dep-count
        >= MIN_REVERSE_DEPS whose def-use chain extends past the chunk boundary).
        At each chunk close, compute INPUTS (vars referenced inside, defined
        outside, in chunk order) and OUTPUTS (vars defined inside, used after).
        If a chunk's input or output arity exceeds MAX_TUPLE_ARITY, force a
        chunk split EARLIER. Returns chunks in deterministic sequence order."""
    ```
