```python
# New module-level state — extracted once per file
import re
import sys
from pathlib import Path

# Per D-20 policy classes
F64_CONST_PATTERN = re.compile(r'^\s*(?:pub\s+)?const\s+([A-Z][A-Z0-9_]*)\s*:\s*f64\s*=')
F32_CONST_PATTERN = re.compile(r'^\s*(?:pub\s+)?const\s+([A-Z][A-Z0-9_]*)\s*:\s*f32\s*=')

# Markers for non-generic context (file lacks `<F: Float>` anywhere in body)
GENERIC_F_FN_PATTERN = re.compile(r'fn\s+\w+\s*<\s*F\s*:\s*Float\s*>')

def classify_file(path: Path) -> dict:
    """Return per-file metadata for the classifier."""
    src = path.read_text()
    lines = src.split('\n')
    f64_consts = set()
    f32_consts = set()
    for line in lines:
        m = F64_CONST_PATTERN.match(line)
        if m: f64_consts.add(m.group(1))
        m = F32_CONST_PATTERN.match(line)
        if m: f32_consts.add(m.group(1))
    is_generic = bool(GENERIC_F_FN_PATTERN.search(src))
    return {
        'src': src,
        'f64_consts': f64_consts,
        'f32_consts': f32_consts,
        'is_generic': is_generic,
    }

def classify_site(ident: str, meta: dict, line_context: str) -> str:
    """Classify a single F::new(IDENT) site. Returns the replacement spelling.
    
    Per D-20 policy:
    - f64 const  -> F::cast_from(IDENT)
    - f32 const  -> F::new(IDENT)  (keep, accepted by Float::new)
    - doc-comment / string-literal context -> bare IDENT (revert)
    - non-generic file -> bare IDENT (revert; file should have full revert)
    - numeric token (handled by existing _wrap_f64_literals, not this classifier)
    """
    # Doc-comment context (line starts with `///`, `//!`, or `//`)
    stripped = line_context.lstrip()
    if stripped.startswith('///') or stripped.startswith('//!') or stripped.startswith('//'):
        return ident  # revert
    
    # String-literal context: look for unbalanced `"` to the left of F::new(IDENT)
    # This is a coarse but safe heuristic; Serena's preview will catch edge cases
    quote_count = line_context.count('"')
    if quote_count % 2 != 0:  # odd → currently inside a string literal
        return ident  # revert
    
    # Non-generic file (e.g. deferred.rs) — should already be fully reverted in Task 1
    if not meta['is_generic']:
        return ident  # revert
    
    # f64 const → cast_from (architectural fix for the 11-06 HALT root cause)
    if ident in meta['f64_consts']:
        return f"F::cast_from({ident})"
    
    # f32 const → keep F::new (Float::new(val: f32) accepts f32)
    if ident in meta['f32_consts']:
        return f"F::new({ident})"
    
    # Unknown identifier — likely a function param or local var, leave bare
    # (avoiding spurious wraps; F::new only takes f32 literals/values, F::cast_from takes CubePrimitive values)
    return ident  # revert
```

# After classify_file(path), but before existing transformations
F_NEW_IDENT_PATTERN = re.compile(r'F::new\(([A-Z][A-Z0-9_]+)\)')

def apply_cast_from_policy(src: str, meta: dict) -> str:
    """Apply D-20 cast_from classifier to every F::new(IDENT) site (multi-char IDENT)."""
    lines = src.split('\n')

        def repl(m):
            ident = m.group(1)
            return classify_site(ident, meta, line)
        new_line = F_NEW_IDENT_PATTERN.sub(repl, line)
        out_lines.append(new_line)
    return '\n'.join(out_lines)
