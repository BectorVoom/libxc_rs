D-01 regularization helper (mirror prepare.rs EXACTLY — copied here so the canary is self-contained):
```rust
fn regularize_inputs(rho: &[f64], sigma: &[f64], tau: &[f64]) -> (Vec<f64>, Vec<f64>) {
    let dens_threshold = 1e-15_f64; let tau_threshold = 1e-20_f64;
    let sigma_threshold = dens_threshold.powf(4.0 / 3.0);
    let sigma_floor = sigma_threshold * sigma_threshold;
    let mut s = Vec::with_capacity(sigma.len()); let mut t = Vec::with_capacity(tau.len());
    for i in 0..rho.len() {
        if rho[i] < dens_threshold { s.push(sigma[i]); t.push(tau[i]); continue; }
        let my_rho = dens_threshold.max(rho[i]);
        let my_sigma = sigma_floor.max(sigma[i]);
        let my_tau = tau_threshold.max(tau[i]);
        s.push(my_sigma.min(8.0 * my_rho * my_tau)); t.push(my_tau);
    }
    (s, t)
}
```
The oracle gets RAW inputs (libxc regularizes internally); the Rust launch gets the regularized
(sigma, tau). They must agree at 1e-12.
